import json
from pathlib import Path
from uuid import uuid4

import jev
from questions import QUESTIONS, FLOOR, gate

# hermes wrote the intent. jev answers it. code decides.
def desk(intent, state):
    answer = jev.ask(state, questions=QUESTIONS)
    reduces = answer.choice in ("close", "flatten")
    ok = answer.confidence >= FLOOR and (answer.risk_ok or reduces)
    dest = "execute" if ok and gate(answer) else "review"
    if answer.choice == "hold":
        dest = "skip"

    folder = Path(__file__).parent / "queue" / dest
    folder.mkdir(parents=True, exist_ok=True)
    job = folder / f"{uuid4().hex[:8]}.json"
    job.write_text(json.dumps(dict(
        intent=intent, choice=answer.choice,
        size=answer.size, conf=answer.confidence,
    )))
    return job

# hard rules live in code, not in the model.
from pathlib import Path
from jev import choice, score, flag

QUESTIONS = {
    "action": choice("buy", "sell", "hold", "close", "flatten"),
    "size": score(0.5, 1, 2, 4),            # % of capital
    "risk_ok": flag("spread, latency, loss, feed age"),
}

FLOOR = 0.85                  # below this a human looks first
DAILY_LOSS_CAP = 2000         # usd, then the desk stops
MAX_POSITION = 0.04           # 4% of capital
KILL_FILE = Path("KILL")      # one file halts the desk

def gate(answer):
    if KILL_FILE.exists() or answer.daily_pnl <= -DAILY_LOSS_CAP:
        return False
    return answer.size / 100 <= MAX_POSITION
